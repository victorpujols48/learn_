WITH
juegos_largos AS (
  SELECT title, developer, playtime_hours
  FROM videogames
  WHERE playtime_hours > 50
),
conteo_devs AS (
  SELECT developer, COUNT(*) AS total
  FROM videogames
  GROUP BY developer
)
SELECT j.title, j.playtime_hours, c.total
FROM juegos_largos j
JOIN conteo_devs c ON j.developer = c.developer;
